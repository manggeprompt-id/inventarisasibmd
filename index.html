Script SQL Supabase — SIMBMD DIKPORA & Sekolah

Salin & jalankan pada menu SQL Editor di Supabase sebelum mengubah IS_PREVIEW = false

-- SCRIPT SQL SUPABASE: SIMBMD DIKPORA & SATUAN PENDIDIKAN (SEKOLAH)
CREATE TABLE IF NOT EXISTS public.satuan_pendidikan (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  npsn_username TEXT UNIQUE NOT NULL,
  password TEXT NOT NULL DEFAULT 'dikpora2026',
  nama_sekolah TEXT NOT NULL,
  jenjang TEXT DEFAULT 'SD',
  kepala_sekolah TEXT,
  nama_operator TEXT,
  status_aktif BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public.kamus_kode_barang (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  kode_barang TEXT UNIQUE NOT NULL,
  nama_barang TEXT NOT NULL,
  kategori_kib TEXT NOT NULL DEFAULT 'KIB B - Peralatan & Mesin',
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public.aset_bmd (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  kode_barang TEXT NOT NULL,
  nup INT NOT NULL DEFAULT 1,
  nama_barang TEXT NOT NULL, -- Nama Barang Acuan dari Kamus Excel
  nama_jenis_barang TEXT NOT NULL, -- Spesifikasi / Jenis Barang yang diketik user
  kategori_kib TEXT NOT NULL,
  tahun_perolehan INT NOT NULL DEFAULT 2026,
  nilai_perolehan NUMERIC NOT NULL DEFAULT 0,
  kondisi TEXT NOT NULL DEFAULT 'Baik',
  status_keberadaan TEXT NOT NULL DEFAULT 'Ada',
  ruangan_lokasi TEXT,
  pemegang_barang TEXT,
  sekolah_id UUID REFERENCES public.satuan_pendidikan(id) ON DELETE SET NULL,
  keterangan TEXT,
  updated_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public.log_inventarisasi (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  aset_id UUID REFERENCES public.aset_bmd(id) ON DELETE CASCADE,
  tanggal_cek DATE NOT NULL DEFAULT CURRENT_DATE,
  petugas_nama TEXT NOT NULL,
  kondisi_awal TEXT,
  kondisi_baru TEXT NOT NULL,
  keberadaan TEXT NOT NULL DEFAULT 'Ada',
  catatan_lapangan TEXT,
  nomor_bap TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE IF NOT EXISTS public.mutasi_bmd (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  aset_id UUID REFERENCES public.aset_bmd(id) ON DELETE CASCADE,
  jenis_mutasi TEXT NOT NULL,
  opd_asal TEXT NOT NULL,
  opd_tujuan TEXT NOT NULL,
  tanggal_mutasi DATE NOT NULL DEFAULT CURRENT_DATE,
  nomor_sk TEXT NOT NULL,
  status_persetujuan TEXT NOT NULL DEFAULT 'Menunggu',
  created_at TIMESTAMPTZ DEFAULT now()
);

ALTER TABLE public.satuan_pendidikan ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.kamus_kode_barang ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.aset_bmd ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.log_inventarisasi ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.mutasi_bmd ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public Access satuan_pendidikan" ON public.satuan_pendidikan FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public Access kamus_kode_barang" ON public.kamus_kode_barang FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public Access aset_bmd" ON public.aset_bmd FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public Access log_inventarisasi" ON public.log_inventarisasi FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Public Access mutasi_bmd" ON public.mutasi_bmd FOR ALL USING (true) WITH CHECK (true);
